def get_target_files(input_dir: str, file_filter: str = None):
    all_files = [f for f in os.listdir(input_dir) if os.path.isfile(os.path.join(input_dir, f))]
    
    if not file_filter:
        return [os.path.join(input_dir, f) for f in all_files]

    # New Logic: If the filter looks like a specific list of files (e.g., "img1.jpg,img2.jpg")
    if "," in file_filter or file_filter in all_files:
        target_list = [f.strip() for f in file_filter.split(",")]
        return [os.path.join(input_dir, f) for f in target_list if f in all_files]
        
    # Standard Glob/Regex logic
    import fnmatch
    return [os.path.join(input_dir, f) for f in all_files if fnmatch.fnmatch(f, file_filter)]
	
def parse_simple_filter(user_input):
    # Regex to find: [only: KEY OPERATOR VALUE]
    match = re.search(r'\[only:\s*(\w+)\s+(is|above|below|has)\s+(.*?)\]', user_input, re.IGNORECASE)
    
    if match:
        key, operator, value = match.groups()
        
        # Simple Translation Logic
        ops = {"is": "==", "above": ">", "below": "<", "has": "in"}
        py_op = ops.get(operator.lower(), "==")
        
        # Return a structured filter that your internal common method can use
        return {"key": key.lower(), "op": py_op, "val": value.strip()}
    return None
	
def parse_multi_filter(user_input):
    # 1. Find the filter block
    block_match = re.search(r'\[filter:\s*(.*?)\]', user_input, re.IGNORECASE)
    if not block_match:
        return None
    
    raw_filters = block_match.group(1).split(",")
    parsed_conditions = []

    for item in raw_filters:
        # 2. Extract Key, Operator (is/above/below), and Value
        item_match = re.search(r'(\w+)\s+(is|above|below)\s+(.*)', item.strip(), re.IGNORECASE)
        if item_match:
            key, op, val = item_match.groups()
            parsed_conditions.append({
                "key": key.lower().strip(),
                "op": op.lower().strip(),
                "val": val.lower().strip()
            })
            
    return parsed_conditions # Returns a list of condition dicts
